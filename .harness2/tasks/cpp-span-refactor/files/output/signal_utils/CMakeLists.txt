cmake_minimum_required(VERSION 3.20)
project(signal_utils LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(signal_utils src/signal_utils.cpp)
target_include_directories(signal_utils PUBLIC include)
enable_testing()
find_package(GTest REQUIRED)
add_executable(test_signal_utils tests/test_signal_utils.cpp)
target_link_libraries(test_signal_utils PRIVATE signal_utils GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(test_signal_utils)
